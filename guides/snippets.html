ב-Controller

public IActionResult Index(int page = 1)
{
    int pageSize = 9;
    int total = _context.Products.Count();

    var products = _context.Products
        .OrderBy(p => p.Id)
        .Skip((page - 1) * pageSize)
        .Take(pageSize)
        .ToList();

    ViewBag.Page = page;
    ViewBag.TotalPages = (int)Math.Ceiling(total / (double)pageSize);
    return View(products);
}

יצירת המנהל הראשון

ב-Program.cs (דורש .AddRoles<IdentityRole>())

using (var scope = app.Services.CreateScope())
{
    var roleManager = scope.ServiceProvider.GetRequiredService<RoleManager<IdentityRole>>();
    var userManager = scope.ServiceProvider.GetRequiredService<UserManager<IdentityUser>>();

    if (!await roleManager.RoleExistsAsync("Admin"))
    {
        await roleManager.CreateAsync(new IdentityRole("Admin"));
    }

    var admin = await userManager.FindByEmailAsync("admin@example.com");
    if (admin != null && !await userManager.IsInRoleAsync(admin, "Admin"))
    {
        await userManager.AddToRoleAsync(admin, "Admin");
    }
}

⭐ רק הנתונים של המשתמש המחובר ("ההזמנות שלי")

כל משתמש רואה רק את מה שהוא יצר. מתאים להזמנות, פוסטים, משימות.

במודל: שדה שישמור מי יצר

public string UserId { get; set; } = "";

ב-Controller (בראש הקובץ: using System.Security.Claims;)

[Authorize]
public IActionResult MyOrders()
{
    var userId = User.FindFirstValue(ClaimTypes.NameIdentifier);
    var orders = _context.Orders.Where(o => o.UserId == userId).ToList();
    return View(orders);
}

[HttpPost]
[ValidateAntiForgeryToken]
[Authorize]
public IActionResult Create(Order order)
{
    order.UserId = User.FindFirstValue(ClaimTypes.NameIdentifier)!;
    ModelState.Remove("UserId");

    if (!ModelState.IsValid)
    {
        return View(order);
    }
    _context.Orders.Add(order);
    _context.SaveChanges();
    return RedirectToAction("MyOrders");
}

ModelState.Remove("UserId"): את השדה הזה ממלאים בקוד ולא בטופס, אז לא בודקים אותו. ואחרי הוספת השדה למודל: Migration.

בסוף העמוד

@section Scripts {
    <script>
        document.getElementById("showPassword").addEventListener("change", function () {
            var input = document.querySelector('input[type="password"], input.password-field');
            input.classList.add("password-field");
            input.type = this.checked ? "text" : "password";
        });
    </script>
}
